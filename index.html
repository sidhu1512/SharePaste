// SharePaste Zero-Datastore RFC 3986 Pipeline
import { compressZstd } from '@sharepaste/wasm';

export async function generateZeroTrustLink(source: string): Promise<string> {
  // 1. Compress raw UTF-8 bytes in-memory
  const rawBuffer = new TextEncoder().encode(source);
  const compressed = await compressZstd(rawBuffer, { level: 19 });
  
  // 2. Encode to URL-safe Base64 payload
  const fragment = encodeUrlSafeBase64(compressed);
  
  // 3. RFC 3986: Hash fragment is NEVER transmitted over HTTP
  return `https://sidhu1512.github.io/SharePaste/app.html#${fragment}`;
}

Why Zero-Datastore Outperforms Traditional Cloud Pastebins

A staff engineer's guide to how client-side hashing eliminates storage leaks, server exploits, and surveillance vectors.

01

The URL Hash (#) Is Never Transmitted

According to the RFC 3986 URI specification, fragment identifiers (everything following the # character) are strictly client-side constructs. Your web browser strips the fragment before issuing HTTP GET requests to any web server or CDN.

This means our hosting servers, network edge routers, proxies, and ISPs never see your paste contents. Even with a subpoena or database breach, there is literally nothing to hand over.

02

Adaptive Multi-Density QR Routing

Traditional tools try to force massive URLs into unreadable, dense QR codes with tiny pixel modules that fail to scan under normal camera lenses.

SharePaste continuously checks the compressed URL length against our scannable threshold (350 characters / ≥2.6 px per module). Short snippets render an elegant QR modal immediately; longer snippets automatically pivot to a 1-click clipboard link to guarantee 100% scan reliability.
